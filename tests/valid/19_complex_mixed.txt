int i = 0, total = 0;
for (i = 1; i <= 5; i = i + 1) {
    switch (i) {
        case 1:
            if (total == 0) {
                total = total + 10;
            }
            break;
        case 2:
            total = total + 20;
            break;
        default:
            total = total + 1;
            break;
    }
}
